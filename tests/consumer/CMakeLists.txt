cmake_minimum_required(VERSION 3.31)
project(alpaca_consumer LANGUAGES CXX)
if(ALPACA_SOURCE_DIR)
    include(FetchContent)
    FetchContent_Declare(alpaca_cpp SOURCE_DIR "${ALPACA_SOURCE_DIR}")
    FetchContent_MakeAvailable(alpaca_cpp)
    if(ALPACA_MARKETS_BUILD_TESTS OR ALPACA_MARKETS_BUILD_EXAMPLES)
        message(FATAL_ERROR "Embedded SDK must disable examples and tests by default")
    endif()
else()
    find_package(alpaca_markets 0.2.0 EXACT CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE alpaca::markets)
if(README_EXAMPLES_DIR)
    file(GLOB examples "${README_EXAMPLES_DIR}/*.cpp")
    foreach(example IN LISTS examples)
        get_filename_component(name "${example}" NAME_WE)
        add_executable("${name}" "${example}")
        target_link_libraries("${name}" PRIVATE alpaca::markets)
    endforeach()
endif()
